date +%d - display the date 'DD' format.
[bavan@DESKTOP-CF9TBQU ~]$ date +%d
24

date +%m - display the month in 'MM' format.
[bavan@DESKTOP-CF9TBQU ~]$ date +%m
03

date +%y - display the year in 'YY' format.
[bavan@DESKTOP-CF9TBQU ~]$ date +%y
25

date +%D - display the date in 'MM/DD/YY' format.
[bavan@DESKTOP-CF9TBQU ~]$ date +%D
03/24/25

date +%Y - display the year in 'YYYY' format.
[bavan@DESKTOP-CF9TBQU ~]$ date +%Y
2025

date +%B - display the month in word format.
[bavan@DESKTOP-CF9TBQU ~]$ date +%B
March

date +%b - display the month in word format.
[bavan@DESKTOP-CF9TBQU ~]$ date +%b
Mar

date +%A - display the day in word format.
[bavan@DESKTOP-CF9TBQU ~]$ date +%A
Monday

date +%a - display the day in word format.
[bavan@DESKTOP-CF9TBQU ~]$ date +%a
Mon

date +%h - display the month in word format.
[bavan@DESKTOP-CF9TBQU ~]$ date +%h
Mar

//Example 02
Read two number from user and print 
addition, subtraction, multiplication and divition.

[bavan@DESKTOP-CF9TBQU ~]$ vi example02.sh

echo "Enter a"
read a
echo "Enter number b"
read b

sum=$(($a+$b))
sub=$(($a-$b))
mul=$(($a*$b))
div=$(($a/$b))

echo "Summation	" $sum
echo "Subtraction	" $sub
echo "Multiplication	" $mul
echo "Division	" $div

[bavan@DESKTOP-CF9TBQU ~]$ ls

[bavan@DESKTOP-CF9TBQU ~]$ chmod 777 example02.sh
